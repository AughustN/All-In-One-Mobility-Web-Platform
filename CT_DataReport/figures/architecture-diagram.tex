    \quad ├── Location Search (TomTom API) \\
    \quad ├── Camera Image Proxy \\
    \quad ├── User Authentication (JWT) \\
    \quad └── History Management \\[0.3cm]

    \quad ├── TomTom API (Routing \& Geocoding) \\

    \quad ├── HCMC Traffic Server (Camera Images) \\
    \quad └── MySQL Database (Persistence)
};

\end{tikzpicture}
