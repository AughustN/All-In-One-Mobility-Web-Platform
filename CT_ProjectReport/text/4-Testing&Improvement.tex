\section{Limitations: What is Missing?}
% ============================================

\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.4}
    \begin{tabularx}{\textwidth}{|p{3.5cm}|p{4cm}|X|}
        \hline
        \textbf{Limitation} & \textbf{Impact} & \textbf{Root Cause} \\
        \hline
        No real-time bus data & Cannot filter in-active bus routes & BusMap does not provide public API \\
        \hline
        Static congestion thresholds & Fixed thresholds may not adapt to special events (holidays, accidents) & Thresholds set manually, no ML adaptation \\
        \hline
        Limited place database & AI may suggest places not in database & Database has $\sim$700 places, not comprehensive \\
        \hline
        Camera coverage gaps & Some major roads have no cameras, creating blind spots & Dependent on public camera availability \\
        \hline
        Walking routes approximate & Uses straight-line distance estimation, not actual walking paths & TomTom pedestrian API not integrated \\
        \hline
        No Metro integration & Cannot suggest Metro Line 1 routes & Data not yet available at development time \\
        \hline
    \end{tabularx}
    \caption{System Limitations and Root Causes}
\end{table}

\section{Future Improvements: What Would We Add Next?}
% ============================================

\subsection{Priority 1: Real-time Transit Data}
\begin{itemize}[noitemsep]
    \item \textbf{Bus GPS Integration}: Partner with HCMC bus operator to access real-time bus locations
    \item \textbf{Metro Line 1 Support}: Add metro stations and schedules.
    \item \textbf{Arrival Predictions}: Use historical data + ML to predict actual arrival times
\end{itemize}

\subsection{Priority 2: Enhanced AI Capabilities}
\begin{itemize}[noitemsep]
    \item \textbf{Expanded Place Database}: Web scraping to add more restaurants, cafes, attractions
    \item \textbf{Personalized Recommendations}: Learn user preferences from history
    \item \textbf{Voice Input}: Add speech-to-text for hands-free queries
\end{itemize}

\subsection{Priority 3: Improved Detection}
\begin{itemize}[noitemsep]
    \item \textbf{Dynamic Thresholds}: ML-based congestion classification that adapts over time
    \item \textbf{Incident Detection}: Detect accidents, road closures from camera feeds
    \item \textbf{Traffic Violation Detection}: Red light running, wrong-way driving alerts
\end{itemize}

\subsection{Priority 4: User Experience}
\begin{itemize}[noitemsep]
    \item \textbf{Offline Maps}: Download map tiles and bus routes for offline use
    \item \textbf{Push Notifications}: Alerts for congestion on saved routes
    \item \textbf{Multi-modal Routing}: Combine bus + walking + bike-sharing in one trip
    \item \textbf{Social Features}: Share trips with friends, collaborative planning
\end{itemize}

\subsection{Priority 5: Infrastructure}
\begin{itemize}[noitemsep]
    \item \textbf{Load Balancing}: Handle more concurrent users with horizontal scaling
    \item \textbf{API Rate Limiting}: Prevent abuse and ensure fair usage
    \item \textbf{Monitoring Dashboard}: Real-time metrics for API health and performance
    \item \textbf{Mobile App}: Native iOS/Android apps for better performance
\end{itemize}
